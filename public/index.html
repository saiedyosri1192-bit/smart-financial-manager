2. بناء قاعدة البيانات المحاسبية الموسعة بشجرة الحسابات ونظام العمولات والرقابة def init_db(): conn = sqlite3.connect('accounting_system.db') cursor = conn.cursor() # جدول اليومية العامة الشامل (يشمل الضريبة، الفني المسؤول، والعمولات المقررة) cursor.execute('''CREATE TABLE IF NOT EXISTS journal_entries ( entry_id INTEGER PRIMARY KEY AUTOINCREMENT, date TEXT, description TEXT, debit_account TEXT, credit_account TEXT, amount REAL, tax_amount REAL, party_name TEXT, payment_status TEXT, due_date TEXT, technician_name TEXT, commission_amount REAL )''') # جدول فنيي التركيبات والمهندسين (الرواتب، بدلات المواقع، والعمولات التراكمية) cursor.execute('''CREATE TABLE IF NOT EXISTS payroll ( emp_id INTEGER PRIMARY KEY AUTOINCREMENT, emp_name TEXT, role TEXT, base_salary REAL, allowances REAL, deductions REAL, total_commissions REAL DEFAULT 0.0 )''') # جدول مستودع بضاعة الكاميرات والأجهزة الأمنية وملحقاتها (شاشات، كابلات، NVR/DVR) cursor.execute('''CREATE TABLE IF NOT EXISTS inventory ( prod_id INTEGER PRIMARY KEY AUTOINCREMENT, product_name TEXT, quantity INTEGER, cost_price REAL, selling_price REAL )''') # جدول إدارة عهد الفنيين النقدية لشراء المستلزمات الطارئة للمواقع والمشاريع cursor.execute('''CREATE TABLE IF NOT EXISTS petty_cash ( cash_id INTEGER PRIMARY KEY AUTOINCREMENT, employee_name TEXT, amount_given REAL, amount_spent REAL, remaining_balance REAL, purpose TEXT )''') # جدول الأمان وإدارة الصلاحيات cursor.execute('''CREATE TABLE IF NOT EXISTS users ( username TEXT PRIMARY KEY, password_hash TEXT, role TEXT )''') # سجل العمليات الرقابي لـ مراجع الحسابات لمنع التلاعب وتتبع كل كبيرة وصغيرة (Audit Trail) cursor.execute('''CREATE TABLE IF NOT EXISTS system_logs ( log_id INTEGER PRIMARY KEY AUTOINCREMENT, timestamp TEXT, username TEXT, action_type TEXT, description TEXT )''') #
