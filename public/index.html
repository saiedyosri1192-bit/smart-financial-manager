========================================== # 1. محرك الحسابات وقاعدة البيانات والذكاء المحاسبي # ========================================== DB_NAME = 'financial_manager.db' class EnterpriseCoreEngine: def __init__(self, db_name=DB_NAME): self.db_name = db_name self.init_db() def get_connection(self): conn = sqlite3.connect(self.db_name) conn.row_factory = sqlite3.Row return conn def init_db(self): with self.get_connection() as conn: cursor = conn.cursor() # محرك ترحيل وإدخال القيود المزدوجة مع التوازن def post_journal_entry(self, entry_date, description, reference, lines): total_debit = sum(float(l['debit']) for l in lines) total_credit = sum(float(l['credit']) for l in lines) if abs(total_debit - total_credit) > 0.001: raise ValueError(f"القيد غير متوازن! إجمالي المدين ({total_debit:,.2f}) لا يساوي الدائن ({total_credit:,.2f})") with self.get_connection() as conn: cursor = conn.cursor() cursor.execute(''' INSERT INTO journal_entries (entry_date, description, reference, posted_by, status) VALUES (?, ?, ?, 'المحاسب', 'معتمد') ''', (entry_date, description, reference)) entry_id = cursor.lastrowid for l in lines: code = l['account_code'] debit = float(l['debit']) credit = float(l['credit']) cursor.execute('INSERT INTO journal_lines (entry_id, account_code, debit, credit) VALUES (?, ?, ?, ?)', (entry_id, code, debit, credit)) # تحديث رصيد الحساب تلقائياً في شجرة الحسابات net_change = debit - credit cursor.execute('UPDATE coa SET balance = balance + ? WHERE code = ?', (net_change, code)) conn.commit() return entry_id # حذف قيد محاسبي وإعادة تصحيح الأرصدة def delete_journal_entry(self, entry_id): with self.get_connection() as conn: cursor = conn.cursor() lines = cursor.execute('SELECT account_code, debit, credit FROM journal_lines WHERE entry_id = ?', (entry_id,)).fetchall() for line in lines: net_change = line['debit'] - line['credit'] cursor.execute('UPDATE coa SET balance = balance - ? WHERE code = ?', (net_change, line['account_code'])) cursor.execute('DELETE FROM journal_entries WHERE id = ?', (entry_id,)) conn.commit() # محرك استخراج البيانات من الصور والملفات المرفقة (OCR / PDF / Excel) def parse_external_file(self, file_path, filename): ext = filename.split('.')[-1].lower() extracted_text = "" if ext in ['xlsx', 'xls', 'csv']: df = pd.read_excel(file_path) if 'xls' in ext else pd.read_csv(file_path) return {"type": "excel", "rows": len(df), "data": df.to_dict(orient='records')} elif ext == 'pdf' and HAS_PDF: with pdfplumber.open(file_path) as pdf: for page in pdf.pages: extracted_text += (page.extract_text() or "") + "\n" return {"type": "pdf", "text": extracted_text} elif ext in ['png', 'jpg', 'jpeg'] and HAS_OCR: img = Image.open(file_path) extracted_text = pytesseract.image_to_string(img, lang='ara+eng') return {"type": "ocr", "text": extracted_text} return {"type": "unknown", "text": "تم حفظ الملف بنجاح. لا تتطابق مكتبة المعالجة التلقائية حالياً."} # ========================================== # 2. تطبيق خادم FLASK مع الربط بالواجهة # ========================================== app = Flask(__name__) engine = EnterpriseCoreEngine() @app.route('/') def index(): conn = engine.get_connection() coa = conn.execute('SELECT * FROM coa ORDER BY code ASC').fetchall() entries = conn.execute(''' SELECT j.*, SUM(l.debit) as total_debit FROM journal_entries j LEFT JOIN journal_lines l ON j.id = l.entry_id GROUP BY j.id ORDER BY j.id DESC ''').fetchall() projects = conn.execute('SELECT * FROM projects').fetchall() maintenance = conn.execute('SELECT * FROM maintenance').fetchall() techs = conn.execute('SELECT * FROM tech_custody').fetchall() # قراءة الكود المصمم واقترانه ببيانات السيرفر with open(__file__, 'r', encoding='utf-8') as f: code_content = f.read() # فصل قالب الـ HTML المدمج في الجزء السفلي html_part = code_content.split('"""HTML_TEMPLATE_MARKER"""')[1] return render_template_string(html_part, coa=coa, entries=entries, projects=projects, maintenance=maintenance, techs=techs) #
