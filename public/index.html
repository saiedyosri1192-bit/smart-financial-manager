========================================== # 1. المحرك المحاسبي (Accounting Engine) # ========================================== DB_NAME = 'financial_manager.db' class AccountingEngine: def __init__(self, db_name=DB_NAME): self.db_name = db_name self.init_db() def get_connection(self): conn = sqlite3.connect(self.db_name) conn.row_factory = sqlite3.Row return conn def init_db(self): with self.get_connection() as conn: cursor = conn.cursor() # 1. شجرة الحسابات (COA) cursor.execute(''' CREATE TABLE IF NOT EXISTS coa ( code TEXT PRIMARY KEY, name TEXT NOT NULL, account_type TEXT NOT NULL, balance REAL DEFAULT 0.0 ) ''') # 2. قيود اليومية العامة cursor.execute(''' CREATE TABLE IF NOT EXISTS journal_entries ( id INTEGER PRIMARY KEY AUTOINCREMENT, entry_date TEXT NOT NULL, description TEXT NOT NULL, posted_by TEXT NOT NULL, is_posted INTEGER DEFAULT 0 ) ''') # 3. تفاصيل بنود القيد cursor.execute(''' CREATE TABLE IF NOT EXISTS journal_lines ( id INTEGER PRIMARY KEY AUTOINCREMENT, entry_id INTEGER NOT NULL, account_code TEXT NOT NULL, debit REAL DEFAULT 0.0, credit REAL DEFAULT 0.0, FOREIGN KEY(entry_id) REFERENCES journal_entries(id), FOREIGN KEY(account_code) REFERENCES coa(code) ) ''') # إدخال شجرة الحسابات الافتراضية default_accounts = [ ('110100', 'الصندوق الرئيسي / النقدية', 'ASSET', 50000.0), ('110200', 'بنك الراجحي الرئيسي', 'ASSET', 180000.0), ('110300', 'العملاء والذمم المدينة', 'ASSET', 45000.0), ('110501', 'مخزون الكاميرات والأجهزة', 'ASSET', 65000.0), ('110600', 'ضريبة المدخلات المستردة 15%', 'ASSET', 6000.0), ('210100', 'الموردون والذمم الدائنة', 'LIABILITY', 25000.0), ('210200', 'ضريبة المخرجات المستحقة 15%', 'LIABILITY', 10500.0), ('310000', 'رأس المال المعتمد', 'EQUITY', 200000.0), ('410100', 'إيرادات المبيعات والتركيبات', 'REVENUE', 130000.0), ('510100', 'تكلفة البضاعة المباعة (COGS)', 'COGS', 72000.0), ('610100', 'المصروفات التشغيلية', 'EXPENSE', 5500.0) ] for code, name, acc_type, bal in default_accounts: cursor.execute('INSERT OR IGNORE INTO coa VALUES (?, ?, ?, ?)', (code, name, acc_type, bal)) conn.commit() def post_journal_entry(self, description, lines, posted_by='CFO'): total_debit = sum(float(line.get('debit', 0)) for line in lines) total_credit = sum(float(line.get('credit', 0)) for line in lines) if abs(total_debit - total_credit) > 0.001: raise ValueError(f"القيد غير متوازن! إجمالي المدين: {total_debit} - إجمالي الدائن: {total_credit}") date_str = datetime.now().strftime('%Y-%m-%d %H:%M:%S') with self.get_connection() as conn: cursor = conn.cursor() cursor.execute(''' INSERT INTO journal_entries (entry_date, description, posted_by, is_posted) VALUES (?, ?, ?, 1) ''', (date_str, description, posted_by)) entry_id = cursor.lastrowid for line in lines: debit = float(line.get('debit', 0)) credit = float(line.get('credit', 0)) code = line['account_code'] cursor.execute(''' INSERT INTO journal_lines (entry_id, account_code, debit, credit) VALUES (?, ?, ?, ?) ''', (entry_id, code, debit, credit)) net_change = debit - credit cursor.execute('UPDATE coa SET balance = balance + ? WHERE code = ?', (net_change, code)) conn.commit() return entry_id def import_excel_data(self, file_path): df = pd.read_excel(file_path) return df.to_dict(orient='records') def parse_pdf_text(self, file_path): if not HAS_PDF: return "مكتبة pdfplumber غير مثبة، يرجى تشغيل: pip install pdfplumber" text = "" with pdfplumber.open(file_path) as pdf: for page in pdf.pages: text += (page.extract_text() or "") return text # ========================================== # 2. واجهة المستخدم HTML # ========================================== HTML_TEMPLATE = """ المنظومة المحاسبية الموحدة ERP

""" # ========================================== # 3. خادم التطبيق (Flask Web Server) # ========================================== app = Flask(__name__) engine = AccountingEngine() @app.route('/') def
