function submitManualEntry() { const date = document.getElementById('entryDate').value; const desc = document.getElementById('entryDesc').value; const ref = document.getElementById('entryRef').value; const debitCode = document.getElementById('debitCode').value.trim(); const debitAmt = parseFloat(document.getElementById('debitAmt').value) || 0; const creditCode = document.getElementById('creditCode').value.trim(); const creditAmt = parseFloat(document.getElementById('creditAmt').value) || 0; if (!desc || debitAmt <= 0 || creditAmt <= 0) { alert('يرجى كتابة البيان وتحديد مبالغ مدين ودائن أكبر من الصفر!'); return; } if (debitAmt !== creditAmt) { alert('خطأ توازن: المبلغ المدين ينبغي أن يطابق المبلغ الدائن بالضبط!'); return; } const payload = { date: date, description: desc, reference: ref, lines: [ { account_code: debitCode, debit: debitAmt, credit: 0 }, { account_code: creditCode, debit: 0, credit: creditAmt } ] }; fetch('/api/add_entry', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }) .then(res => res.json()) .then(data => { if (data.status === 'success') { alert('تم الترحيل وتحديث الأرصدة بنجاح!'); location.reload(); } else { alert('فشل الترحيل: ' + data.message); } }) .catch(err => { alert('حدث خطأ في الاتصال بالسيرفر أثناء الترحيل'); }); }
