import os import sys import sqlite3 import unittest from datetime import datetime from flask import Flask, render_template_string, request, jsonify, send_file import pandas as pd import pdfplumber # ========================================== # 1. المكون الأقوى: المحرك المحاسبي (Accounting Engine) # ========================================== DB_NAME = 'financial_manager.db' class AccountingEngine: def __init__(self, db_name=DB_NAME): self.db_name = db_name self.init_db() def get_connection(self): conn = sqlite3.connect(self.db_name) conn.row_factory = sqlite3.Row return conn def init_db(self): with self.get_connection() as conn: cursor = conn.cursor() # 1. شجرة الحسابات (COA) cursor.execute(''' CREATE TABLE IF NOT EXISTS coa ( code TEXT PRIMARY KEY, name TEXT NOT NULL, account_type TEXT NOT NULL, balance REAL DEFAULT 0.0 ) ''') # 2. قيود اليومية العامة cursor.execute(''' CREATE TABLE IF NOT EXISTS journal_entries ( id INTEGER PRIMARY KEY AUTOINCREMENT, entry_date TEXT NOT NULL, description TEXT NOT NULL, posted_by TEXT NOT NULL, is_posted INTEGER DEFAULT 0 ) ''') # 3. تفاصيل بنود القيد (مدين/دائن) cursor.execute(''' CREATE TABLE IF NOT EXISTS journal_lines ( id INTEGER PRIMARY KEY AUTOINCREMENT, entry_id INTEGER NOT NULL, account_code TEXT NOT NULL, debit REAL DEFAULT 0.0, credit REAL DEFAULT 0.0, FOREIGN KEY(entry_id) REFERENCES journal_entries(id), FOREIGN KEY(account_code) REFERENCES coa(code) ) ''') # 4. جدول الفواتير الضريبية (ZATCA) cursor.execute(''' CREATE TABLE IF NOT EXISTS invoices ( id TEXT PRIMARY KEY, invoice_type TEXT NOT NULL, party_name TEXT NOT NULL, invoice_date TEXT NOT NULL, subtotal REAL NOT NULL, vat_amount REAL NOT NULL, total_amount REAL NOT NULL, status TEXT DEFAULT 'POSTED' ) ''') # إدخال شجرة الحسابات الافتراضية للحلول الأمنية default_accounts = [ ('110100', 'الصندوق الرئيسي / النقدية', 'ASSET', 50000.0), ('110200', 'بنك الراجحي الرئيسي', 'ASSET', 180000.0), ('110300', 'العملاء والذمم المدينة', 'ASSET', 45000.0), ('110501', 'مخزون الكاميرات والأجهزة', 'ASSET', 65000.0), ('110600', 'ضريبة المدخلات المستردة 15%', 'ASSET', 6000.0), ('210100', 'الموردون والذمم الدائنة', 'LIABILITY', 25000.0), ('210200', 'ضريبة المخرجات المستحقة 15%', 'LIABILITY', 10500.0), ('310000', 'رأس المال المعتمد', 'EQUITY', 200000.0), ('410100', 'إيرادات المبيعات والتركيبات', 'REVENUE', 130000.0), ('510100', 'تكلفة البضاعة المباعة (COGS)', 'COGS', 72000.0), ('610100', 'المصروفات التشغيلية', 'EXPENSE', 5500.0) ] for code, name, acc_type, bal in default_accounts: cursor.execute('INSERT OR IGNORE INTO coa VALUES (?, ?, ?, ?)', (code, name, acc_type, bal)) conn.commit() def post_journal_entry(self, description, lines, posted_by='CFO'): """مراجعة وترحيل القيد المحاسبي المتوازن 100%""" total_debit = sum(float(line.get('debit', 0)) for line in lines) total_credit = sum(float(line.get('credit', 0)) for line in lines) if abs(total_debit - total_credit) > 0.001: raise ValueError(f"القيد غير متوازن! إجمالي المدين: {total_debit} - إجمالي الدائن: {total_credit}") date_str = datetime.now().strftime('%Y-%m-%d %H:%M:%S') with self.get_connection() as conn: cursor = conn.cursor() cursor.execute(''' INSERT INTO journal_entries (entry_date, description, posted_by, is_posted) VALUES (?, ?, ?, 1) ''', (date_str, description, posted_by)) entry_id = cursor.lastrowid for line in lines: debit = float(line.get('debit', 0)) credit = float(line.get('credit', 0)) code = line['account_code'] cursor.execute(''' INSERT INTO journal_lines (entry_id, account_code, debit, credit) VALUES (?, ?, ?, ?) ''', (entry_id, code, debit, credit)) # تحديث أرصدة شجرة الحسابات تلقائياً net_change = debit - credit cursor.execute('UPDATE coa SET balance = balance + ? WHERE code = ?', (net_change, code)) conn.commit() return entry_id def import_excel_data(self, file_path): df = pd.read_excel(file_path) return df.to_dict(orient='records') def parse_pdf_text(self, file_path): text = "" with pdfplumber.open(file_path) as pdf: for page in pdf.pages: text += (page.extract_text() or "") return text # ========================================== # 2. الواجهة الرسومية HTML الشاملة (Integrated Template) # ========================================== HTML_TEMPLATE = """ المنظومة المحاسبية والمالية الموحدة - Enterprise ERP

منظومة العملاق المحاسبية الشاملة

الربط التلقائي بين المدير المالي، ومدير الحسابات، والمحاسب، والمراجع

تصدير COA Excel

يقوم المحرك بقراءة وتفريغ المستندات آلياً داخل شجرة الحسابات والدفاتر

إدخال وترحيل قيد محاسبي جديد

شجرة الحسابات والديدات (COA)

{% for acc in coa %} {% endfor %}
الرمزالحسابالرصيد
{{ acc['code'] }} {{ acc['name'] }} {{ "{:,.2f}".format(acc['balance']) }}

دفتر اليومية العامة والقيود المرحّلة

{% for j in journals %} {% endfor %}
القيدالتاريخالبيانالمبلغ
JV-{{ j['id'] }} {{ j['entry_date'] }} {{ j['description'] }} {{ "{:,.2f}".format(j['total_amount']) if j['total_amount'] else '0.00' }}
""" # ========================================== # 3. تطبيق الويب الخادم (Flask App Server) # ========================================== app = Flask(__name__) engine = AccountingEngine() @app.route('/') def dashboard(): conn = engine.get_connection() coa = conn.execute('SELECT * FROM coa').fetchall() journals = conn.execute(''' SELECT j.id, j.entry_date, j.description, j.posted_by, SUM(l.debit) as total_amount FROM journal_entries j LEFT JOIN journal_lines l ON j.id = l.entry_id GROUP BY j.id ORDER BY j.id DESC ''').fetchall() return render_template_string(HTML_TEMPLATE, coa=coa, journals=journals) @app.route('/api/post_entry', methods=['POST']) def post_entry(): data = request.json try: entry_id = engine.post_journal_entry( description=data['description'], lines=data['lines'], posted_by='CFO' ) return jsonify({'status': 'success', 'entry_id': entry_id}) except Exception as e: return jsonify({'status': 'error', 'message': str(e)}), 400 @app.route('/api/export_excel', methods=['GET']) def export_excel(): conn = engine.get_connection() df = pd.read_sql_query('SELECT * FROM coa', conn) file_path = 'COA_Financial_Export.xlsx' df.to_excel(file_path, index=False) return send_file(file_path, as_attachment=True) @app.route('/api/import_file', methods=['POST']) def import_file(): if 'file' not in request.files: return jsonify({'status': 'error', 'message': 'لم يتم رفع ملف'}), 400 file = request.files['file'] filename = file.filename.lower() os.makedirs('uploads', exist_ok=True) file_path = os.path.join('uploads', file.filename) file.save(file_path) if filename.endswith('.xlsx') or filename.endswith('.xls'): parsed_data = engine.import_excel_data(file_path) return jsonify({'status': 'success', 'type': 'excel', 'data': parsed_data}) elif filename.endswith('.pdf'): parsed_text = engine.parse_pdf_text(file_path) return jsonify({'status': 'success', 'type': 'pdf', 'raw_text': parsed_text}) return jsonify({'status': 'error', 'message': 'الصيغة غير مدعومة'}), 400 # ========================================== # 4. وحدة الاختبارات الآلية (Test Suite) # ========================================== class TestAccountingEngine(unittest.TestCase): def setUp(self): self.test_db = 'test_financial.db' self.engine = AccountingEngine(db_name=self.test_db) def tearDown(self): if os.path.exists(self.test_db): os.remove(self.test_db) def test_balanced_entry(self): lines = [ {'account_code': '110100', 'debit': 1000.0, 'credit': 0.0}, {'account_code': '410100', 'debit': 0.0, 'credit': 1000.0} ] entry_id = self.engine.post_journal_entry("قيد مبيعات متوازن", lines) self.assertIsNotNone(entry_id) def test_unbalanced_entry(self): lines = [ {'account_code': '110100', 'debit': 1000.0, 'credit': 0.0}, {'account_code': '410100', 'debit': 0.0, 'credit': 500.0} ] with self.assertRaises(ValueError): self.engine.post_journal_entry("قيد خاطئ", lines) # ========================================== # 5. نقطة التشغيل الرئيسية (Main Execution) # ========================================== if __name__ == '__main__': # تشغيل الفحص الآلي والتأكد من سلامة الكود أولاً suite = unittest.TestLoader().loadTestsFromTestCase(TestAccountingEngine) runner = unittest.TextTestRunner(verbosity=0) result = runner.run(suite) if result.wasSuccessful(): print("=== تم اجتياز الاختبارات المحاسبية بنجاح 100% ===") print("=== جاري تشغيل الخادم المحاسبي الموحد على http://127.0.0.1:5000 ===") app.run(debug=True, port=5000) else: print("خطأ في الاختبارات المحاسبية!")
